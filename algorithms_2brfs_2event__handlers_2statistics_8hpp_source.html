220 void record_parallel_beam_chunk(size_t chunk_size,
221 std::chrono::nanoseconds worker_compute_time,
222 std::chrono::nanoseconds main_thread_merge_time,
223 std::chrono::nanoseconds main_thread_intern_time,
224 std::chrono::nanoseconds fluent_slot_time,
225 std::chrono::nanoseconds numeric_slot_time,
226 std::chrono::nanoseconds derived_slot_time,
227 std::chrono::nanoseconds state_lookup_time,
228 std::chrono::nanoseconds reached_atom_update_time)
229 {
230 ++m_num_parallel_beam_chunk_flushes;
231 m_num_parallel_beam_chunk_tasks_total += chunk_size;
232 m_max_parallel_beam_chunk_size = std::max<uint64_t>(m_max_parallel_beam_chunk_size, chunk_size);
233 m_parallel_beam_worker_compute_time_ns += static_cast<uint64_t>(worker_compute_time.count());
234 m_parallel_beam_main_thread_merge_time_ns += static_cast<uint64_t>(main_thread_merge_time.count());
235 m_parallel_beam_main_thread_intern_time_ns += static_cast<uint64_t>(main_thread_intern_time.count());
236 m_parallel_beam_fluent_slot_time_ns += static_cast<uint64_t>(fluent_slot_time.count());
237 m_parallel_beam_numeric_slot_time_ns += static_cast<uint64_t>(numeric_slot_time.count());
238 m_parallel_beam_derived_slot_time_ns += static_cast<uint64_t>(derived_slot_time.count());
239 m_parallel_beam_state_lookup_time_ns += static_cast<uint64_t>(state_lookup_time.count());
240 m_parallel_beam_reached_atom_update_time_ns += static_cast<uint64_t>(reached_atom_update_time.count());
241 }
