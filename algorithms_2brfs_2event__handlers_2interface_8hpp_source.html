129template<typename Derived_>
130class EventHandlerBase : public IEventHandler
131{
132protected:
133 Statistics m_statistics;
134 formalism::Problem m_problem;
135 bool m_quiet;
136
137private:
138 EventHandlerBase() = default;
139 friend Derived_;
140
142 constexpr const auto& self() const { return static_cast<const Derived_&>(*this); }
143 constexpr auto& self() { return static_cast<Derived_&>(*this); }
144
145public:
146 explicit EventHandlerBase(formalism::Problem problem, bool quiet = true) : m_statistics(), m_problem(problem), m_quiet(quiet) {}
147
148 void on_expand_state(const State& state) override
149 {
150 m_statistics.increment_num_expanded();
151
152 if (!m_quiet)
153 {
154 self().on_expand_state_impl(state);
155 }
156 }
157
158 void on_expand_goal_state(const State& state) override
159 {
160 m_statistics.increment_num_expanded_goal_states();
161
162 if (!m_quiet)
163 {
164 self().on_expand_goal_state_impl(state);
165 }
166 }
167
168 void on_generate_state(const State& state, formalism::GroundAction action, ContinuousCost action_cost, const State& successor_state) override
169 {
170 m_statistics.increment_num_generated();
171
172 if (!m_quiet)
173 {
174 self().on_generate_state_impl(state, action, action_cost, successor_state);
175 }
176 }
177
178 /* Computing a novelty witness costs a scan of the transition's novel atoms, so a search only
179 pays for it when a handler says it consumes the witness. That is a capability of the concrete
180 handler, not of its verbosity: a non-quiet handler that never reads witnesses (this base's own
181 `on_generate_state_with_novel_witness` ignores them) would otherwise make every search compute
182 them for nothing, and a handler that does want them while staying quiet could not ask. Derived
183 handlers that implement `on_generate_state_with_novel_witness` override this to return true. */
184 bool supports_novel_witness_events() const override { return false; }
185
186 void on_generate_state_with_novel_witness(const State& state,
187 formalism::GroundAction action,
188 ContinuousCost action_cost,
189 const State& successor_state,
190 const iw::AtomIndexList& novel_fluent_atom_indices) override
191 {
192 [[maybe_unused]] const auto& ignored_state = state;
193 [[maybe_unused]] const auto ignored_action = action;
194 [[maybe_unused]] const auto ignored_action_cost = action_cost;
195 [[maybe_unused]] const auto& ignored_successor_state = successor_state;
196 [[maybe_unused]] const auto& ignored_novel_fluent_atom_indices = novel_fluent_atom_indices;
197 }
198
199 bool supports_payloadless_generated_state_events() const override { return m_quiet; }
200
201 /* The payloadless hooks feed exactly the same counters as their payloadful counterparts, so a
202 search that reports a transition without payload is indistinguishable in the statistics from
203 one that reports it with payload. */
204 void on_generate_state_without_payload() override { m_statistics.increment_num_generated(); }
205
206 void on_generate_state_in_search_tree_without_payload() override { m_statistics.increment_num_generated_in_search_tree(); }
207
208 void on_generate_state_not_in_search_tree_without_payload() override { m_statistics.increment_num_generated_not_in_search_tree(); }
209
210 void on_generate_state_in_search_tree(const State& state, formalism::GroundAction action, ContinuousCost action_cost, const State& successor_state) override
211 {
212 m_statistics.increment_num_generated_in_search_tree();
213
214 if (!m_quiet)
215 {
216 self().on_generate_state_in_search_tree_impl(state, action, action_cost, successor_state);
217 }
218 }
219
220 void
221 on_generate_state_not_in_search_tree(const State& state, formalism::GroundAction action, ContinuousCost action_cost, const State& successor_state) override
222 {
223 m_statistics.increment_num_generated_not_in_search_tree();
224
225 if (!m_quiet)
226 {
227 self().on_generate_state_not_in_search_tree_impl(state, action, action_cost, successor_state);
228 }
229 }
230
231 void on_finish_g_layer(DiscreteCost g_value) override
232 {
233 m_statistics.on_finish_g_layer(static_cast<int64_t>(g_value));
234
235 if (!m_quiet)
236 {
237 assert(!m_statistics.get_num_expanded_until_g_value().empty());
238 self().on_finish_g_layer_impl(g_value, m_statistics.get_num_expanded_until_g_value().back(), m_statistics.get_num_generated_until_g_value().back());
239 }
240 }
241
242 void on_start_search(const State& start_state) override
243 {
244 m_statistics = Statistics();
245
246 m_statistics.set_search_start_time_point(std::chrono::high_resolution_clock::now());
247
248 if (!m_quiet)
249 {
250 self().on_start_search_impl(start_state);
251 }
252 }
253
254 void on_end_search(uint64_t num_reached_fluent_atoms,
255 uint64_t num_reached_derived_atoms,
256 uint64_t num_states,
257 uint64_t num_nodes,
258 uint64_t num_actions,
259 uint64_t num_axioms) override
260 {
261 m_statistics.set_search_end_time_point(std::chrono::high_resolution_clock::now());
262 m_statistics.set_num_reached_fluent_atoms(num_reached_fluent_atoms);
263 m_statistics.set_num_reached_derived_atoms(num_reached_derived_atoms);
264 m_statistics.set_num_states(num_states);
265 m_statistics.set_num_nodes(num_nodes);
266 m_statistics.set_num_actions(num_actions);
267 m_statistics.set_num_axioms(num_axioms);
268
269 if (!m_quiet)
270 {
271 self().on_end_search_impl(num_reached_fluent_atoms, num_reached_derived_atoms, num_states, num_nodes, num_actions, num_axioms);
272 }
273 }
274
275 void on_finish_iw1_incremental_first_applicability(
276 const IW1IncrementalFirstApplicabilityStatistics& iw1_incremental_first_applicability_statistics) override
277 {
278 m_statistics.set_iw1_incremental_first_applicability_statistics(iw1_incremental_first_applicability_statistics);
279 }
280
281 void on_finish_parallel_beam_chunk(size_t chunk_size,
282 std::chrono::nanoseconds worker_compute_time,
283 std::chrono::nanoseconds main_thread_merge_time,
284 std::chrono::nanoseconds main_thread_intern_time,

287 std::chrono::nanoseconds derived_slot_time,
288 std::chrono::nanoseconds state_lookup_time,

290 {
291 m_statistics.record_parallel_beam_chunk(chunk_size,
292 worker_compute_time,
293 main_thread_merge_time,
294 main_thread_intern_time,
295 fluent_slot_time,
296 numeric_slot_time,
297 derived_slot_time,
298 state_lookup_time,
299 reached_atom_update_time);
300 }
301
302 void on_finish_parallel_beam_pipeline(size_t ready_queue_high_water,
303 size_t in_flight_chunks_high_water,
304 std::chrono::nanoseconds consumer_stall_time,
305 std::chrono::nanoseconds producer_stall_time) override
306 {
307 m_statistics.record_parallel_beam_pipeline(ready_queue_high_water,
308 in_flight_chunks_high_water,
309 consumer_stall_time,
310 producer_stall_time);
311 }
312
313 void on_solved(const Plan& plan) override
314 {
315 if (!m_quiet)
316 {
317 self().on_solved_impl(plan);
318 }
319 }
320
321 void on_unsolvable() override
322 {
323 if (!m_quiet)
324 {
325 self().on_unsolvable_impl();
326 }
327 }
328
329 void on_exhausted() override
330 {
331 if (!m_quiet)
332 {
333 self().on_exhausted_impl();
334 }
335 }
336
338 const Statistics& get_statistics() const override { return m_statistics; }
339 bool is_quiet() const { return m_quiet; }
340};
